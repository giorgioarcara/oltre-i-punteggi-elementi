\node[block] (A3) at (13.2,0) {\textbf{Confronto con soglia}};

\node[example, below=0.3cm of A1] (A1ex) {\textit{es. punteggio = 20}};
\node[example, below=0.3cm of A2] (A2ex) {\textit{es. prestazione al di sotto di 3\% risultati peggiori}};
